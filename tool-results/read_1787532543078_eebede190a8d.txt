    76→
    77→export function ProfileView() {
    78→  const { navigate } = useNav();
    79→  const { user, setUser, logout } = useAuth();
    80→  const { toast } = useToast();
    81→  const [copied, setCopied] = useState(false);
    82→  const [editOpen, setEditOpen] = useState(false);
    83→  const [editName, setEditName] = useState(user?.name || "");
    84→  const [editAvatar, setEditAvatar] = useState(user?.avatar || "");
    85→  const [savingEdit, setSavingEdit] = useState(false);
    86→  const [verifyingAdult, setVerifyingAdult] = useState(false);
    87→
    88→  // Estado de suscripción fresco
    89→  const { data: subData, isLoading: subLoading } = useQuery<SubStatusResponse>({
    90→    queryKey: ["subscription-status"],
    91→    queryFn: () => api.get<SubStatusResponse>("/subscriptions/status"),
    92→    staleTime: 30 * 1000,
    93→  });
    94→
    95→  // Historial de pagos
    96→  const { data: paymentsData, isLoading: paymentsLoading } =
    97→    useQuery<PaymentsResponse>({
    98→      queryKey: ["payments"],
    99→      queryFn: () => api.get<PaymentsResponse>("/payments?limit=20"),
   100→      staleTime: 30 * 1000,
   101→    });
   102→
   103→  if (!user) {
   104→    return (
   105→      <div className="flex min-h-screen flex-col bg-background">
   106→        <AppHeader />
   107→        <main className="flex flex-1 flex-col items-center justify-center px-4 pb-20 pt-20 text-center">
   108→          <p className="text-muted-foreground">
   109→            Necesitás iniciar sesión para ver tu perfil.
   110→          </p>
   111→          <Button className="mt-4" onClick={() => navigate("login")}>
   112→            Iniciar sesión
   113→          </Button>
   114→        </main>
   115→      </div>
   116→    );
   117→  }
   118→
   119→  const sub = subData?.subscription;
   120→  const isActive = subData?.active === true;
   121→  const daysLeft = sub?.daysLeft ?? 0;

   125→    const plan = sub.plan;
   126→    const totalDays = plan.durationDays || 30;
   127→    const elapsed = Math.max(0, totalDays - daysLeft);
   128→    const pct = totalDays > 0 ? (elapsed / totalDays) * 100 : 0;
   129→    return Math.min(100, Math.max(0, Math.round(pct)));
   130→  })();

   132→  const payments = paymentsData?.items || [];
   133→

   135→    const token = user.token || user.accessToken || "";

   137→    try {
   138→      await navigator.clipboard.writeText(token);
   139→      setCopied(true);
   140→      setTimeout(() => setCopied(false), 1800);
   141→      toast({ title: "Token copiado ✓" });
   142→    } catch {
   143→      toast({
   144→        title: "No se pudo copiar",
   145→        description: "Copiá el token manualmente.",
   146→        variant: "destructive",
   147→      });
   148→    }
   149→  };
   150→
   151→  const handleLogout = () => {
   152→    logout();
   153→    setUser(null);
   154→    toast({ title: "Sesión cerrada" });
   155→    navigate("landing");
   156→  };
   157→
   158→  const handleSaveEdit = async () => {
   159→    const trimmedName = editName.trim();
   160→    if (trimmedName.length < 2) {
   161→      toast({
   162→        title: "Nombre inválido",
   163→        description: "El nombre debe tener al menos 2 caracteres.",
   164→        variant: "destructive",
   165→      });
   166→      return;
   167→    }
   168→    setSavingEdit(true);
   169→    try {
   170→      const trimmedAvatar = editAvatar.trim();
   171→      const res = await api.put<{ user: AuthUser }>("/auth/me", {
   172→        name: trimmedName,
   173→        avatar: trimmedAvatar || null,
   174→      });
   175→      setUser(res.user);
   176→      setSavingEdit(false);
   177→      setEditOpen(false);
   178→      toast({ title: "Perfil actualizado ✓" });
   179→    } catch (e: any) {
   180→      setSavingEdit(false);
   181→      const message =
   182→        e instanceof ApiError ? e.message : "No se pudo actualizar el perfil";
   183→      toast({
   184→        title: "Error al guardar",
   185→        description: message,
   186→        variant: "destructive",
   187→      });
   188→    }
   189→  };
   190→
   191→  const handleVerifyAdult = async () => {
   192→    setVerifyingAdult(true);
   193→    try {
   194→      const res = await apiFetch<{ user: AuthUser }>("/auth/me", {
   195→        method: "PATCH",
   196→        body: JSON.stringify({ adultVerified: true }),
   197→      });
   198→      setUser(res.user);
   199→      toast({ title: "Verificación +18 activada ✓" });
   200→    } catch (e: any) {
   201→      const message =
   202→        e instanceof ApiError ? e.message : "No se pudo verificar";
   203→      toast({
   204→        title: "Error",
   205→        description: message,
   206→        variant: "destructive",
   207→      });
   208→    } finally {
   209→      setVerifyingAdult(false);
   210→    }
   211→  };
   212→
   213→  return (
   214→    <div className="flex min-h-screen flex-col bg-background">
   215→      <AppHeader />
   216→      <main className="flex-1">
   217→        <div className="view-enter mx-auto max-w-4xl px-4 pb-20 pt-20 sm:px-6 md:pb-8 lg:px-8">
   218→          {/* Title */}
   219→          <motion.div
   220→            initial={{ opacity: 0, y: -10 }}
   221→            animate={{ opacity: 1, y: 0 }}
   222→            transition={{ duration: 0.3 }}
   223→            className="mb-6"
   224→          >
   225→            <h1 className="flex items-center gap-2 text-2xl font-black sm:text-3xl">
   226→              <UserIcon className="h-6 w-6 text-primary" />
   227→              <span className="text-gradient-red">Mi Perfil</span>
   228→            </h1>
   229→            <p className="mt-1 text-sm text-muted-foreground">
   230→              Gestioná tu cuenta, suscripción y métodos de acceso.
   231→            </p>
   232→          </motion.div>
   233→
   234→          {/* User info card */}
   235→          <motion.div
   236→            initial={{ opacity: 0, y: 10 }}
   237→            animate={{ opacity: 1, y: 0 }}
   238→            transition={{ duration: 0.3, delay: 0.05 }}
   239→          >
   240→            <Card className="card-hover border-border/60 bg-card/60 backdrop-blur">
   241→              <CardContent className="flex flex-col items-center gap-5 p-5 sm:flex-row sm:p-6">
   242→                <Avatar className="h-20 w-20 border-2 border-primary/40 sm:h-24 sm:w-24">
   243→                  {user.avatar ? (
   244→                    <AvatarImage src={user.avatar} alt={user.name} />
   245→                  ) : null}
   246→                  <AvatarFallback className="bg-primary/20 text-2xl font-black text-primary sm:text-3xl">
   247→                    {user.name?.charAt(0).toUpperCase() || "U"}
   248→                  </AvatarFallback>
   249→                </Avatar>
   250→
   251→                <div className="flex-1 text-center sm:text-left">
   252→                  <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   253→                    <h2 className="text-xl font-bold text-white sm:text-2xl">
   254→                      {user.name}
   255→                    </h2>
   256→                    {user.role === "ADMIN" && (
   257→                      <Badge className="bg-yellow-600/90 text-white">
   258→                        <Shield className="mr-1 h-3 w-3" /> Admin
   259→                      </Badge>
   260→                    )}
   261→                    {isActive && (
   262→                      <Badge className="bg-primary/90 text-primary-foreground">
   263→                        <Crown className="mr-1 h-3 w-3" /> Premium
   264→                      </Badge>
   265→                    )}
   266→                    {!user.adultVerified && (
   267→                      <Badge variant="outline" className="border-yellow-600/50 text-yellow-500">
   268→                        <ShieldAlert className="mr-1 h-3 w-3" /> No verificado +18
   269→                      </Badge>
   270→                    )}
   271→                  </div>
   272→                  <div className="mt-2 flex flex-col items-center gap-1 text-sm text-muted-foreground sm:flex-row sm:items-center sm:gap-3">
   273→                    <span className="flex items-center gap-1.5">
   274→                      <Mail className="h-3.5 w-3.5" /> {user.email}
   275→                    </span>
   276→                  </div>
   277→                  <div className="mt-1 text-xs text-muted-foreground/80">
   278→                    Miembro desde{" "}
   279→                    {user.subscription?.endDate
   280→                      ? formatDate(user.subscription.endDate)
   281→                      : "recientemente"}
   282→                  </div>
   283→                </div>
   284→
   285→                <div className="flex gap-2 sm:flex-col">
   286→                  <Button
   287→                    variant="outline"
   288→                    size="sm"
   289→                    onClick={() => {
   290→                      setEditName(user.name);
   291→                      setEditAvatar(user.avatar || "");
   292→                      setEditOpen(true);
   293→                    }}
   294→                  >
   295→                    <Edit2 className="mr-1.5 h-3.5 w-3.5" />
   296→                    Editar
   297→                  </Button>
   298→                  <Button
   299→                    variant="outline"
   300→                    size="sm"
   301→                    onClick={handleLogout}
   302→                    className="text-destructive hover:text-destructive"
   303→                  >
   304→                    <LogOut className="mr-1.5 h-3.5 w-3.5" />
   305→                    Salir
   306→                  </Button>
   307→                </div>
   308→              </CardContent>
   309→            </Card>
   310→          </motion.div>
   311→
   312→          {/* Grid: subscription + token */}
   313→          <div className="mt-4 grid gap-4 md:grid-cols-2">
   314→            {/* Subscription status */}
   315→            <motion.div
   316→              initial={{ opacity: 0, y: 10 }}
   317→              animate={{ opacity: 1, y: 0 }}
   318→              transition={{ duration: 0.3, delay: 0.1 }}
   319→            >
   320→              <Card
   321→                className={`h-full border-border/60 bg-card/60 backdrop-blur ${
   322→                  isActive ? "glow-red-sm card-hover" : "card-hover"
   323→                }`}
   324→              >
   325→                <CardHeader>
   326→                  <CardTitle className="flex items-center gap-2 text-base">
   327→                    <Crown className="h-4 w-4 text-primary" />
   328→                    Suscripción
   329→                  </CardTitle>
   330→                </CardHeader>
   331→                <CardContent>
   332→                  {subLoading ? (
   333→                    <div className="space-y-2">
   334→                      <Skeleton className="h-5 w-1/2" />
   335→                      <Skeleton className="h-4 w-3/4" />
   336→                      <Skeleton className="h-2 w-full" />
   337→                    </div>
   338→                  ) : isActive && sub ? (
   339→                    <div className="space-y-3">
   340→                      <div className="flex items-center justify-between">
   341→                        <div>
   342→                          <p className="text-lg font-bold text-white">
   343→                            {sub.plan?.name || "Plan"}
   344→                          </p>
   345→                          <p className="text-xs text-muted-foreground">
   346→                            Estado:{" "}
   347→                            <span className="font-medium text-green-500">
   348→                              {subscriptionStatusLabel(sub.status)}
   349→                            </span>
   350→                          </p>
   351→                        </div>
   352→                        <Badge className="bg-green-600/90 text-white">
   353→                          <Check className="mr-1 h-3 w-3" /> Activa
   354→                        </Badge>
   355→                      </div>
   356→
   357→                      {/* Quality + screens */}
   358→                      <div className="flex flex-wrap gap-2 text-xs">
   359→                        {sub.plan?.quality && (
   360→                          <span className="flex items-center gap-1 rounded bg-primary/15 px-2 py-1 text-primary">
   361→                            <Sparkles className="h-3 w-3" /> {sub.plan.quality}
   362→                          </span>
   363→                        )}
   364→                        {sub.plan?.screens && (
   365→                          <span className="flex items-center gap-1 rounded bg-secondary px-2 py-1 text-secondary-foreground">
   366→                            <Tv className="h-3 w-3" /> {sub.plan.screens} pantallas
   367→                          </span>
   368→                        )}
   369→                      </div>
   370→
   371→                      <Separator />
   372→
   373→                      {/* Days left */}
   374→                      <div>
   375→                        <div className="mb-1 flex items-center justify-between text-xs">
   376→                          <span className="flex items-center gap-1 text-muted-foreground">
   377→                            <Calendar className="h-3 w-3" />
   378→                            Días restantes
   379→                          </span>
   380→                          <span className="font-bold text-white">
   381→                            {daysLeft} / {sub.plan?.durationDays ?? 30}
   382→                          </span>
   383→                        </div>
   384→                        <Progress value={subProgress} className="h-2" />
   385→                        <p className="mt-1 text-[11px] text-muted-foreground">
   386→                          Vence el {formatDate(sub.endDate)}
   387→                        </p>
   388→                      </div>
   389→
   390→                      <Button
   391→                        variant="outline"
   392→                        size="sm"
   393→                        className="w-full"
   394→                        onClick={() => navigate("subscription")}
   395→                      >
   396→                        <Crown className="mr-1.5 h-3.5 w-3.5" />
   397→                        Ver planes / Renovar
   398→                      </Button>
   399→                    </div>
   400→                  ) : (
   401→                    <div className="space-y-3">
   402→                      <div className="flex items-center gap-2 text-muted-foreground">
   403→                        <AlertCircle className="h-4 w-4 text-yellow-500" />
   404→                        <span className="text-sm">Sin suscripción activa</span>
   405→                      </div>
   406→                      <p className="text-xs text-muted-foreground">
   407→                        Suscribite para desbloquear todo el catálogo en la mejor
   408→                        calidad, sin anuncios y en múltiples pantallas.
   409→                      </p>
   410→                      <Button
   411→                        size="sm"
   412→                        className="w-full"
   413→                        onClick={() => navigate("subscription")}
   414→                      >
   415→                        <Crown className="mr-1.5 h-4 w-4" />
   416→                        Suscribirse ahora
   417→                      </Button>
   418→                    </div>
   419→                  )}
   420→                </CardContent>
   421→              </Card>
   422→            </motion.div>
   423→
   424→            {/* Access token */}
   425→            <motion.div
   426→              initial={{ opacity: 0, y: 10 }}
   427→              animate={{ opacity: 1, y: 0 }}
   428→              transition={{ duration: 0.3, delay: 0.15 }}
   429→            >
   430→              <Card className="card-hover h-full border-border/60 bg-card/60 backdrop-blur">
   431→                <CardHeader>
   432→                  <CardTitle className="flex items-center gap-2 text-base">
   433→                    <CreditCard className="h-4 w-4 text-primary" />
   434→                    Token de acceso
   435→                  </CardTitle>
   436→                </CardHeader>
   437→                <CardContent>
   438→                  <p className="mb-3 text-xs text-muted-foreground">
   439→                    Usá este token para acceder sin email ni contraseña desde
   440→                    cualquier dispositivo. Mantenelo privado.
   441→                  </p>
   442→                  <div className="flex items-center gap-2 rounded-lg border border-border bg-muted/40 p-2">
   443→                    <code className="flex-1 truncate font-mono text-sm font-bold tracking-wider text-primary">
   444→                      {user.token || user.accessToken || "—"}
   445→                    </code>
   446→                    <Button
   447→                      size="sm"
   448→                      variant="ghost"
   449→                      onClick={copyToken}
   450→                      className="shrink-0"
   451→                      aria-label="Copiar token"
   452→                    >
   453→                      {copied ? (
   454→                        <Check className="h-4 w-4 text-green-500" />
   455→                      ) : (
   456→                        <Copy className="h-4 w-4" />

   458→                    </Button>
   459→                  </div>
   460→                  <div className="mt-3 rounded-md bg-primary/10 p-2 text-[11px] text-muted-foreground">

   467→            </motion.div>

   469→
   470→          {/* Adult verification note */}
   471→          {!user.adultVerified && (
   472→            <motion.div
   473→              initial={{ opacity: 0 }}
   474→              animate={{ opacity: 1 }}
   475→              transition={{ delay: 0.2 }}
   476→              className="mt-4"
   477→            >
   478→              <Card className="card-hover border-yellow-600/30 bg-yellow-600/5">
   479→                <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center">
   480→                  <ShieldAlert className="h-6 w-6 shrink-0 text-yellow-500" />
   481→                  <div className="flex-1">
   482→                    <p className="text-sm font-semibold text-yellow-500">
   483→                      Verificación de adulto pendiente
   484→                    </p>
   485→                    <p className="mt-1 text-xs text-muted-foreground">
   486→                      Tu cuenta no está verificada como adulto. La sección +18
   487→                      está bloqueada. Confirmá que tenés 18 años o más para
   488→                      desbloquearla.
   489→                    </p>
   490→                  </div>
   491→                  <Button
   492→                    variant="outline"
   493→                    size="sm"
   494→                    onClick={handleVerifyAdult}
   495→                    disabled={verifyingAdult}
   496→                    className="shrink-0 border-yellow-600/50 text-yellow-500 hover:bg-yellow-600/10 hover:text-yellow-400"

   499→                      <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" />

   501→                      <ShieldCheck className="mr-1.5 h-3.5 w-3.5" />

   503→                    {verifyingAdult ? "Verificando..." : "Confirmo que tengo 18+"}
   504→                  </Button>
   505→                </CardContent>
   506→              </Card>
   507→            </motion.div>

   509→

   511→          <motion.div
   512→            initial={{ opacity: 0, y: 10 }}
   513→            animate={{ opacity: 1, y: 0 }}
   514→            transition={{ duration: 0.3, delay: 0.2 }}
   515→            className="mt-4"

   517→            <Card className="card-hover border-border/60 bg-card/60 backdrop-blur">
   518→              <CardHeader>
   519→                <CardTitle className="flex items-center gap-2 text-base">
   520→                  <Coins className="h-4 w-4 text-primary" />
   521→                  Historial de pagos

   535→                      Aún no tenés pagos registrados.
   536→                    </p>
   537→                    <Button
   538→                      variant="outline"
   539→                      size="sm"
   540→                      className="mt-3"
   541→                      onClick={() => navigate("subscription")}
   542→                    >
   543→                      Ver planes

   548→                    {payments.map((p) => (

   551→                        className="flex items-center justify-between gap-3 rounded-lg border border-border/60 bg-background/40 p-3"
   552→                      >
   553→                        <div className="min-w-0 flex-1">

   558→                            <Badge
